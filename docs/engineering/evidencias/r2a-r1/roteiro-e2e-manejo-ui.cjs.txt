// R2A-R1: manejo normal pela UI (Morte no Detalhes do Lote) usa o comando do servidor.
const { chromium } = require('playwright');
const W = Number(process.argv[2]), H = Number(process.argv[3]), BASE = process.argv[4], NOME_AREA = process.argv[5];
const OUT = '/tmp/claude-0/r2a-r1';
const API = 'http://127.0.0.1:3334';
const res = [];
const ok = (nome, cond, extra = '') => res.push(`${cond ? 'PASS' : 'FAIL'} ${BASE} ${nome}${extra ? ' — ' + String(extra).slice(0, 170) : ''}`);
const E = 'cmvwd6au200037d5t9qk00js3', S = 'cmvwd6auy00087d5tqtm2rk09';
(async () => {
  const { token } = await (await fetch(`${API}/auth/login`, { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify({ login: 'demo.operador', senha: 'senha-demo-local-123' }) })).json();
  const call = async (m, u, b) => (await fetch(`${API}${u}`, { method: m, headers: { authorization: `Bearer ${token}`, 'content-type': 'application/json' }, body: b ? JSON.stringify(b) : undefined })).json();
  const areas = await call('GET', '/areas-pastagem?limit=500');
  const PASTO = (Array.isArray(areas) ? areas : areas.items).find((a) => a.nome === NOME_AREA);
  const QTD = 300 + Math.floor(Math.random() * 600);
  const lote = await call('POST', '/lotes', { empresa_id: E, nome: `UI ${BASE}`, quantidade_cabecas: QTD, categoria: 'BOI', sexo: 'Macho', setor_id: S, area_entrada_id: PASTO.id, area_atual_id: PASTO.id, data_entrada: '2026-10-01', motivo_entrada: 'Compra', status: 'Ativo' });

  const browser = await chromium.launch({ args: ['--use-gl=swiftshader', '--enable-unsafe-swiftshader'] });
  const mobile = W < 800;
  const ctx = await browser.newContext({ viewport: { width: W, height: H }, isMobile: mobile, hasTouch: mobile });
  const page = await ctx.newPage();
  const erros = []; page.on('pageerror', (e) => erros.push(e.message.slice(0, 160)));
  const reqs = [];
  page.on('request', (r) => { if (!['GET', 'OPTIONS', 'HEAD'].includes(r.method())) reqs.push({ m: r.method(), u: new URL(r.url()).pathname.replace(/^\/(api|native)/, ''), b: r.postData() || '' }); });
  await page.goto('http://127.0.0.1:5180/', { waitUntil: 'networkidle' });
  await page.locator('input').nth(0).fill('demo.operador');
  await page.locator('input[type=password]').fill('senha-demo-local-123');
  await page.keyboard.press('Enter');
  await page.waitForURL(/mapageral/i, { timeout: 20000 });
  await page.waitForTimeout(6000);
  const tocar = async (loc) => { if (mobile) await loc.tap({ force: true }); else await loc.click({ force: true }); };
  // Marcadores se sobrepõem no zoom mobile: clique no elemento exato, não por coordenada.
  await page.getByText(String(QTD), { exact: true }).first().dispatchEvent('click');
  await page.getByText('Detalhes do Lote').first().waitFor({ timeout: 15000 });
  ok('abriu o lote certo', (await page.getByRole('dialog').last().innerText()).includes(`UI ${BASE}`));
  await tocar(page.getByRole('button', { name: 'Eventos' }).first());
  await tocar(page.getByRole('menuitem', { name: 'Morte' }));
  await page.getByText('Registrar Morte').first().waitFor({ timeout: 10000 });
  const dlg = page.getByRole('dialog').last();
  if (!(await dlg.locator('input[placeholder="0"]').count())) await tocar(dlg.getByRole('button', { name: 'Adicionar Categoria' }));
  await dlg.locator('input[placeholder="0"]').first().fill('7');
  await tocar(dlg.getByText('Selecione a causa').first());
  await tocar(page.getByRole('option').first());
  await page.screenshot({ path: `${OUT}/${BASE}-07-morte-pela-ui.png` });
  await tocar(dlg.getByRole('button', { name: 'Registrar Mortes' }));
  await page.waitForTimeout(4000);
  const lotes = await call('GET', '/lotes?limit=500');
  const depois = (Array.isArray(lotes) ? lotes : lotes.items).find((l) => l.id === lote.id);
  ok('saldo no servidor: QTD → QTD-7', depois?.quantidade_cabecas === QTD - 7, `${QTD} → ${depois?.quantidade_cabecas}`);
  const comandos = reqs.filter((r) => r.m === 'POST' && /comandos\/manejo-lote/.test(r.u));
  ok('morte foi pelo comando do servidor (1 POST)', comandos.length === 1, `${comandos.length} POST; corpo: ${comandos.map((c) => c.b.replace(/"empresa_id":"[^"]*",?/, '').slice(0, 150)).join(' ; ')}`);
  ok('sem POST legado /movimentacoes-mapa', !reqs.some((r) => r.m === 'POST' && r.u === '/movimentacoes-mapa'));
  const patchOper = reqs.filter((r) => r.m === 'PATCH' && /^\/lotes\//.test(r.u) && /quantidade_cabecas|"status"|area_atual_id|categoria/.test(r.b));
  ok('nenhum PATCH /lotes operacional sem estado_esperado', patchOper.every((r) => /estado_esperado/.test(r.b)), `${patchOper.length} PATCH operacionais`);
  ok('nenhum DELETE/PATCH de movimento no navegador', !reqs.some((r) => ['DELETE', 'PATCH'].includes(r.m) && /movimentacoes-/.test(r.u)));
  const movs = await call('GET', '/movimentacoes-mapa?limit=500');
  const mov = (Array.isArray(movs) ? movs : movs.items).find((m) => m.lote_id === lote.id && m.tipo === 'Morte');
  ok('movimento tem proveniência do servidor', mov?.efeito_servidor?.comando === 'manejo_mapa_lote');
  const eleg = await call('GET', `/movimentacoes-mapa/${mov?.id}/elegibilidade-cancelamento`);
  ok('movimento criado pela UI é elegível ao cancelamento seguro', eleg?.elegivel === true, JSON.stringify(eleg?.efeitos?.map((f) => [f.antes?.quantidade_cabecas, f.depois?.quantidade_cabecas])));
  ok('sem erro de página', erros.length === 0, erros.join(' ; '));
  console.log('REQS', reqs.map((r) => r.m + ' ' + r.u).join(', '));
  await page.screenshot({ path: `${OUT}/${BASE}-08-apos-morte-pela-ui.png` });
  console.log(res.join('\n'));
  await browser.close();
})().catch(async (e) => { console.log(res.join('\n')); console.log('ERRO', e.message.slice(0, 300)); process.exit(1); });
