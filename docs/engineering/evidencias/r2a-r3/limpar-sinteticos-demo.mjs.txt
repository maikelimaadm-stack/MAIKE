// Inativa lotes sintéticos de execuções anteriores (banco LOCAL de demonstração) que cobrem rótulos no mobile.
const API = 'http://127.0.0.1:3334';
const { token } = await (await fetch(`${API}/auth/login`, { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify({ login: 'demo.operador', senha: 'senha-demo-local-123' }) })).json();
const h = { authorization: `Bearer ${token}`, 'content-type': 'application/json' };
const l = await (await fetch(`${API}/lotes?limit=500`, { headers: h })).json();
let n = 0;
for (const x of (Array.isArray(l) ? l : l.items).filter((x) => /^(TRANSF|UI|UNIF2?) /.test(x.nome) && x.status === 'Ativo')) {
  const r = await fetch(`${API}/lotes/${x.id}`, { method: 'PATCH', headers: h, body: JSON.stringify({ status: 'Inativo', estado_esperado: { status: 'Ativo' } }) });
  n += r.ok ? 1 : 0;
}
console.log('inativados', n);
