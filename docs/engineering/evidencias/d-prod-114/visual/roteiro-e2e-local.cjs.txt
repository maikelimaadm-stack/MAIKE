    if (!/Boi/.test(await trig.innerText())) await escolher(trig, new RegExp(BOI.replace('+', '\\+')));
    await lin.locator('input[type=number]').first().fill('40');
    await escolher(lin.getByText('Selecione a causa'), 'Doença');
    const m0 = reqs.length;
    await tocar(dlg.getByRole('button', { name: /^Registrar/ }).last());
    await page.waitForTimeout(3000);
    ok('B2 morte dependente: UM POST de baixa', reqs.slice(m0).filter((r) => /comandos\/(baixa-rebanho|manejo-lote)$/.test(r.u)).length === 1, reqs.slice(m0).map((r) => r.u).join(','));
    const [movM] = await movPorTipo('Morte');
    await fechar();
    // Histórico da junção: abre pelo lote PRIN (10 cab após a morte).
    await page.goto(APP + '/MapaGeral', { waitUntil: 'networkidle' });
    const marcadores = page.getByText('10', { exact: true });
    await marcadores.last().waitFor({ timeout: 20000 });
    await marcadores.last().dispatchEvent('click');
    await page.getByText('Detalhes do Lote').first().waitFor({ timeout: 20000 });
    await abrirHistorico();
    const eJ = await esperarEstado('Junção de Lotes', 'Exclusão bloqueada');
    const eM = await esperarEstado('Morte', 'elegivel');
    ok('B2 antes do clique: Junção BLOQUEADA, Morte elegível', eJ === 'Exclusão bloqueada' && eM === 'elegivel', `${eJ} / ${eM}`);
    const bJb = card('Junção de Lotes').getByRole('button', { name: 'Exclusão bloqueada' });
    ok('B2 bloqueado: disabled + aria-describedby + motivo visível', (await bJb.isDisabled()) && !!(await bJb.getAttribute('aria-describedby')) && /dependente/.test(await card('Junção de Lotes').innerText()));
    // POST 409 sem mutação (recusa no servidor, mesma regra do card).
    const fotoAntes = JSON.stringify([await projecao(), await lotesCad(), await recibosExcluidos(), await p.auditLog.count({ where: w })]);
    const r409 = await call('POST', `/movimentacoes-mapa/${movJ}/exclusao`, { operation_id: `x-${TAG}`, motivo: 'tentativa bloqueada' });
    ok('B3 POST com dependência ativa → 409 e ZERO mutação', r409.status === 409 && JSON.stringify([await projecao(), await lotesCad(), await recibosExcluidos(), await p.auditLog.count({ where: w })]) === fotoAntes, `${r409.status} ${r409.j.code}`);
    await page.screenshot({ path: `${OUT}/${BASE}-B2-juncao-bloqueada.png` });
    await excluirPelaUi('Morte', 'Morte lançada errado (E2E sintético)');
    ok('B2 Junção desbloqueou sem recarregar', (await esperarEstado('Junção de Lotes', 'elegivel')) === 'elegivel');
    await excluirPelaUi('Junção de Lotes', 'Junção lançada errado (E2E sintético)');
    const fj = await fatoApagado(movJ);
    ok('B2 junção apagada fisicamente com auditoria', fj.mov === 0 && !fj.ev && fj.aud === 1, JSON.stringify(fj));
    ok('B2 estado pré-junção EXATO (quantidades, pesos, linhagem); Cadastro intacto; 2 recibos excluido', JSON.stringify(await projecao()) === JSON.stringify(preJ.posicoes) && JSON.stringify(await lotesCad()) === JSON.stringify(preJ.lotes) && (await recibosExcluidos()) - preJ.recibos === 2);
    ok('B2 ledger íntegro, conservação, zero órfãos', (await verificar()) && (await conservacao()) && (await orfaos()) === 0);
    ok('B2 sem rolagem horizontal', await semOverflow());
    await page.screenshot({ path: `${OUT}/${BASE}-B2-apos-exclusao.png` });
    await fechar();

    // C: transversais.
    const eleg = await page.evaluate(() => window.__eleg);
    ok(`C3 no máximo 2 GETs de elegibilidade em voo na página (máx=${eleg.max}, total=${eleg.total}, abortados=${eleg.abortados})`, eleg.max <= 2);
    ok('C nenhum DELETE/PATCH de movimento, nenhum /cancelamento', !reqs.some((r) => (['DELETE', 'PATCH'].includes(r.m) && /movimentacoes-/.test(r.u)) || /\/cancelamento$/.test(r.u)));
    ok('C sem erro de página nem console.error', erros.length === 0 && consoleErros.length === 0, [...erros, ...consoleErros].join(' | '));
    ok('C nenhuma resposta 4xx/5xx inesperada (409 do B3 foi pela API, fora da página)', respostas4xx.length === 0, respostas4xx.join(' | '));
  } catch (e) {
    ok('execução', false, e.message);
    await page.screenshot({ path: `${OUT}/${BASE}-ERRO.png` }).catch(() => {});
  }
  await browser.close();
  await helpers.encerrar();
  console.log(res.join('\n'));
})();
