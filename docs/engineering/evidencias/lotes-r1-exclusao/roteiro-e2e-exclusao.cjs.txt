// MAIKE-LOTES-R1 E2E: exclusão do Cadastro após cancelamento seguro (lógica), bloqueio por movimento ativo,
// exclusão física sem histórico, offline sem envio, histórico com indicador de lote excluído.
const { chromium } = require('playwright');
const W = Number(process.argv[2]), H = Number(process.argv[3]), BASE = process.argv[4], PASTO = process.argv[5];
const TAG = `${BASE}-${Date.now().toString(36).slice(-4)}`;
const OUT = '/tmp/claude-0/lx';
const API = 'http://127.0.0.1:3334';
const E = 'cmvwd6au200037d5t9qk00js3', S = 'cmvwd6auy00087d5tqtm2rk09';
const res = [];
const ok = (nome, cond, extra = '') => res.push(`${cond ? 'PASS' : 'FAIL'} ${BASE} ${nome}${extra ? ' — ' + String(extra).slice(0, 170) : ''}`);
(async () => {
  const { token } = await (await fetch(`${API}/auth/login`, { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify({ login: 'demo.operador', senha: 'senha-demo-local-123' }) })).json();
  const raw = (m, u, b) => fetch(`${API}${u}`, { method: m, headers: { authorization: `Bearer ${token}`, 'content-type': 'application/json' }, body: b ? JSON.stringify(b) : undefined });
  const call = async (m, u, b) => { const r = await raw(m, u, b); const j = await r.json().catch(() => ({})); if (!r.ok) throw new Error(`${m} ${u} ${r.status} ${j.code || ''}`); return j; };
  const areas = await call('GET', '/areas-pastagem?limit=500').then((a) => (Array.isArray(a) ? a : a.items));
  const A = areas.find((a) => a.nome === PASTO);
  const novo = (nome, qtd) => call('POST', '/lotes', { empresa_id: E, nome, quantidade_cabecas: qtd, categoria: 'BOI', sexo: 'Macho', setor_id: S, area_entrada_id: A.id, area_atual_id: A.id, data_entrada: '2026-10-01', motivo_entrada: 'Compra', status: 'Ativo' });
  const manejo = (lote, q) => call('POST', '/movimentacoes-mapa/comandos/manejo-lote', { operation_id: `lx-${lote.id}-${q}`, empresa_id: E, tipo: 'Morte', lote_id: lote.id, quantidade_animais: q, data_movimentacao: '2026-10-31T12:00:00.000Z' });
  const excl = await novo(`EXCL ${TAG}`, 500);
  const m = await manejo(excl, 50);
  await call('POST', `/movimentacoes-mapa/${m.movimentacao.id}/cancelamento`, { operation_id: `lx-c-${excl.id}`, motivo: 'Morte sintética lançada por engano' });
  const ativo = await novo(`ATIVO ${TAG}`, 300);
  await manejo(ativo, 10);
  const livre = await novo(`LIVRE ${TAG}`, 120);
  const livre2 = await novo(`OFF ${TAG}`, 60);
  const lista = async () => (await call('GET', '/lotes')).map((l) => l.id);

  const browser = await chromium.launch({ args: ['--use-gl=swiftshader', '--enable-unsafe-swiftshader'] });
  const mobile = W < 800;
  const ctx = await browser.newContext({ viewport: { width: W, height: H }, isMobile: mobile, hasTouch: mobile });
  const page = await ctx.newPage();
  const erros = []; page.on('pageerror', (e) => erros.push(e.message.slice(0, 160)));
  const deletes = [], perigosos = [];
  page.on('request', (r) => {
    const u = new URL(r.url()).pathname;
    if (r.method() === 'DELETE' && /\/lotes\//.test(u)) deletes.push(u.split('/').pop());
    if (['DELETE', 'PATCH'].includes(r.method()) && /movimentacoes-/.test(u)) perigosos.push(`${r.method()} ${u}`);
  });
  await page.goto('http://127.0.0.1:5180/', { waitUntil: 'networkidle' });
  await page.locator('input').nth(0).fill('demo.operador');
  await page.locator('input[type=password]').fill('senha-demo-local-123');
  await page.keyboard.press('Enter');
  await page.waitForURL(/mapageral/i, { timeout: 20000 });
  await page.goto('http://127.0.0.1:5180/CadastroLotes', { waitUntil: 'networkidle' });
  await page.waitForTimeout(3000);
  const semOverflow = async () => !(await page.evaluate(() => document.documentElement.scrollWidth > window.innerWidth + 1));
  const buscar = async (termo) => {
    const campo = page.getByPlaceholder('Buscar lotes…');
    if (!(await campo.count()) || !(await campo.first().isVisible())) await page.getByRole('button', { name: 'Abrir pesquisa' }).first().click();
    await page.getByPlaceholder('Buscar lotes…').fill(termo);
    await page.keyboard.press('Enter');
    await page.waitForTimeout(1200);
  };
  const selecionar = async (termo) => {
    await buscar(termo);
    await page.locator('tr', { hasText: termo }).first().getByRole('checkbox').first().dispatchEvent('click');
    await page.waitForTimeout(600);
  };
  const excluirBotao = () => page.getByRole('button', { name: /^Excluir/ }).first();

  // 1) Movimento ativo: Excluir explica e não confirma
  await selecionar(`ATIVO ${TAG}`);
  await excluirBotao().click();
  await page.getByText(/Há movimentação ativa deste lote no mapa/).first().waitFor({ timeout: 15000 });
  ok('ativo: aviso com o motivo, sem diálogo de confirmação', (await page.getByRole('alertdialog').count()) === 0 && (await page.getByText('Confirmar exclusão').count()) === 0);
  await page.screenshot({ path: `${OUT}/${BASE}-01-bloqueado-movimento-ativo.png` });
  ok('ativo: continua na API, nenhum DELETE', (await lista()).includes(ativo.id) && !deletes.includes(ativo.id));
  await page.keyboard.press('Escape');

  // 2) Histórico só cancelado: confirmação explica e exclusão é lógica
  await selecionar(`EXCL ${TAG}`);
  await excluirBotao().click();
  await page.getByText(/movimentos cancelados ficarão preservados para auditoria/).first().waitFor({ timeout: 15000 });
  ok('só cancelado: confirmação explica a exclusão lógica', true);
  ok('sem rolagem horizontal', await semOverflow());
  await page.screenshot({ path: `${OUT}/${BASE}-02-confirmacao-exclusao-logica.png` });
  await page.getByRole('alertdialog').getByRole('button', { name: 'Excluir' }).click();
  await page.waitForTimeout(2500);
  const porId = await call('GET', `/lotes/${excl.id}`);
  const movs = await call('GET', '/movimentacoes-mapa?limit=500');
  const mv = (Array.isArray(movs) ? movs : movs.items).find((x) => x.id === m.movimentacao.id);
  ok('só cancelado: sai da lista e dos saldos (GET /lotes), identidade preservada por id com excluido_em',
    !(await lista()).includes(excl.id) && !!porId.excluido_em && porId.quantidade_entrada === 500, JSON.stringify({ excluido_em: porId.excluido_em }));
  ok('só cancelado: movimento continua cancelado com motivo e autor', mv?.situacao === 'cancelado' && /engano/.test(mv?.cancelamento?.motivo || '') && !!mv?.cancelamento?.por_login);
  await page.screenshot({ path: `${OUT}/${BASE}-03-apos-exclusao-logica.png` });

  // 3) Sem histórico: exclusão física com a confirmação padrão
  await selecionar(`LIVRE ${TAG}`);
  await excluirBotao().click();
  await page.getByRole('alertdialog').waitFor({ timeout: 15000 });
  const textoDlg = await page.getByRole('alertdialog').innerText();
  ok('sem histórico: confirmação padrão (sem texto de exclusão lógica)', !/movimentos cancelados/.test(textoDlg), textoDlg.replace(/\n+/g, ' ').slice(0, 120));
  await page.getByRole('alertdialog').getByRole('button', { name: 'Excluir' }).click();
  await page.waitForTimeout(2500);
  ok('sem histórico: exclusão física (GET /lotes/:id → 404)', (await raw('GET', `/lotes/${livre.id}`)).status === 404);

  // 4) Offline: nada enviado
  await selecionar(`OFF ${TAG}`);
  await ctx.setOffline(true);
  await page.waitForTimeout(500);
  await excluirBotao().click();
  await page.getByText(/Exclusão de lote só com conexão/).first().waitFor({ timeout: 15000 });
  ok('offline: avisa, sem DELETE e sem confirmação', !deletes.includes(livre2.id) && (await page.getByRole('alertdialog').count()) === 0);
  await ctx.setOffline(false);
  await page.waitForTimeout(800);
  ok('offline: lote continua na API', (await lista()).includes(livre2.id));

  // 5) Mapa: histórico da área com o movimento cancelado e indicador de lote excluído
  await page.goto('http://127.0.0.1:5180/MapaGeral', { waitUntil: 'networkidle' });
  await page.waitForTimeout(6000);
  const box = await page.getByText(PASTO, { exact: true }).first().boundingBox();
  if (mobile) await page.touchscreen.tap(box.x + box.width / 2, box.y + 1); else await page.mouse.click(box.x + box.width / 2, box.y + 1);
  await page.waitForTimeout(3000);
  const item = page.locator('div.border.rounded-lg').filter({ hasText: `EXCL ${TAG}` }).first();
  await item.waitFor({ timeout: 15000 });
  await page.getByText(/Lote excluído do Cadastro — registro mantido para auditoria/).first().waitFor({ timeout: 15000 }).catch(() => {});
  const ti = (await item.innerText()).replace(/\n+/g, ' | ');
  ok('mapa: histórico da área mostra o cancelado com motivo e o indicador de lote excluído', /Cancelado/.test(ti) && /engano/.test(ti) && /Lote excluído do Cadastro/.test(ti), ti.slice(0, 170));
  await item.scrollIntoViewIfNeeded();
  await page.screenshot({ path: `${OUT}/${BASE}-04-historico-lote-excluido.png` });

  ok('DELETE /lotes só dos lotes confirmados (EXCL e LIVRE)', JSON.stringify(deletes.sort()) === JSON.stringify([excl.id, livre.id].sort()), JSON.stringify(deletes));
  ok('nenhum DELETE/PATCH de movimento no navegador', perigosos.length === 0, perigosos.join(','));
  ok('sem erro de página', erros.length === 0, erros.join(' ; '));
  console.log(res.join('\n'));
  await browser.close();
})().catch((e) => { console.log(res.join('\n')); console.log('ERRO', e.message.slice(0, 300)); process.exit(1); });
