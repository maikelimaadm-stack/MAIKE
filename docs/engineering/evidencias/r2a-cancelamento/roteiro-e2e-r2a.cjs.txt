const { chromium } = require('playwright');
const { execFileSync } = require('node:child_process');
const W = Number(process.argv[2]), H = Number(process.argv[3]), TAG = process.argv[4], PASTO = process.argv[5];
const OUT = '/tmp/claude-0/r2a';
const res = [];
const ok = (nome, cond, extra = '') => res.push(`${cond ? 'PASS' : 'FAIL'} ${TAG} ${nome}${extra ? ' — ' + String(extra).slice(0, 170) : ''}`);
const seed = JSON.parse(execFileSync('node', ['/tmp/claude-0/seed-r2a.mjs', TAG, PASTO]).toString().trim().split('\n').pop());
(async () => {
  const browser = await chromium.launch({ args: ['--use-gl=swiftshader', '--enable-unsafe-swiftshader'] });
  const mobile = W < 800;
  const ctx = await browser.newContext({ viewport: { width: W, height: H }, isMobile: mobile, hasTouch: mobile });
  const page = await ctx.newPage();
  const erros = [];
  page.on('pageerror', (e) => erros.push(e.message.slice(0, 160)));
  const perigosos = [];
  const cancelamentos = [];
  page.on('request', (r) => {
    const u = r.url();
    if (['DELETE', 'PATCH'].includes(r.method()) && /movimentacoes-(mapa|estoque|pecuaria)|suplementacao-(eventos|lotes)/.test(u)) perigosos.push(`${r.method()} ${u}`);
    if (r.method() === 'POST' && /\/cancelamento$/.test(u)) cancelamentos.push(u);
  });
  await page.goto('http://127.0.0.1:5180/', { waitUntil: 'networkidle' });
  await page.locator('input').nth(0).fill('demo.operador');
  await page.locator('input[type=password]').fill('senha-demo-local-123');
  await page.keyboard.press('Enter');
  await page.waitForURL(/mapageral/i, { timeout: 20000 });
  await page.waitForTimeout(6000);
  const tocar = async (loc) => { if (mobile) await loc.tap({ force: true }); else await loc.click({ force: true }); };
  const fechar = async () => { for (let i = 0; i < 3; i++) { await page.keyboard.press('Escape'); await page.waitForTimeout(250); } await page.waitForTimeout(400); };
  const abrirArea = async (nome) => {
    const box = await page.getByText(nome, { exact: true }).first().boundingBox();
    const x = box.x + box.width / 2, y = box.y + 1;
    if (mobile) await page.touchscreen.tap(x, y); else await page.mouse.click(x, y);
    await page.waitForTimeout(2500);
  };
  const dlg = () => page.getByRole('dialog').last();
  const txt = async () => (await dlg().innerText().catch(() => '')).replace(/\n+/g, ' | ');
  const semOverflow = async () => !(await page.evaluate(() => document.documentElement.scrollWidth > window.innerWidth + 1));
  const itemDoLote = (nome) => page.locator('div.border.rounded-lg').filter({ hasText: nome }).filter({ has: page.getByRole('button', { name: 'Cancelar movimentação' }) }).first();

  // 1) Morte cancelável, pelo Histórico da Área
  await abrirArea(seed.pasto);
  let t = await txt();
  ok('histórico da área lista a morte ativa com ação de cancelar', new RegExp(`MORTE ${TAG}`).test(t) && /Ativo/.test(t) && (await page.getByRole('button', { name: 'Cancelar movimentação' }).count()) > 0, t.slice(0, 120));
  await page.screenshot({ path: `${OUT}/${TAG}-01-historico-ativo-com-cancelar.png` });
  await tocar(itemDoLote(`MORTE ${TAG}`).getByRole('button', { name: 'Cancelar movimentação' }));
  await page.getByText('450 → 500 cab').first().waitFor({ timeout: 15000 });
  ok('diálogo mostra o efeito calculado no servidor (450 → 500)', true);
  await page.getByLabel('Motivo do cancelamento').fill('Lançamento sintético de teste E2E');
  await page.screenshot({ path: `${OUT}/${TAG}-02-dialogo-efeito-motivo.png` });
  await tocar(page.getByRole('button', { name: 'Confirmar cancelamento' }));
  await page.getByText(/Cancelado em .* por demo\.operador — Motivo: Lançamento sintético/).first().waitFor({ timeout: 15000 });
  t = await txt();
  ok('após cancelar: original continua no histórico, marcado Cancelado com ator e motivo', /Cancelado/.test(t));
  ok('sem rolagem horizontal', await semOverflow());
  await page.screenshot({ path: `${OUT}/${TAG}-03-historico-cancelado.png` });
  const lote = JSON.parse(execFileSync('node', ['-e', `
    const T=(async()=>{const r=await fetch('http://127.0.0.1:3334/auth/login',{method:'POST',headers:{'content-type':'application/json'},body:JSON.stringify({login:'demo.operador',senha:'senha-demo-local-123'})});const {token}=await r.json();
    const l=await (await fetch('http://127.0.0.1:3334/lotes?limit=500',{headers:{authorization:'Bearer '+token}})).json();const lista=Array.isArray(l)?l:l.items;
    console.log(JSON.stringify(lista.filter(x=>x.id==='${seed.morte}'||x.id==='${seed.abate}').map(x=>({id:x.id,q:x.quantidade_cabecas,s:x.status}))));})();`]).toString());
  const lm = lote.find((x) => x.id === seed.morte);
  ok('saldo do lote voltou a 500 no servidor', lm?.q === 500, JSON.stringify(lm));
  await fechar();

  // 2) Registro legado ambíguo (transferência parcial antiga) → explica, não oferece confirmação
  await abrirArea('PASTO 5');
  const botoes = page.getByRole('dialog').last().getByRole('button', { name: 'Cancelar movimentação' });
  if (await botoes.count()) {
    await tocar(botoes.first());
    await page.waitForTimeout(2500);
    t = await txt();
    const confirmar = await page.getByRole('button', { name: 'Confirmar cancelamento' }).count();
    ok('legado ambíguo: explica o bloqueio e não oferece confirmação', /sem vínculo comprovado|registros posteriores|não corresponde|Não há registro/.test(t) && confirmar === 0, t.slice(-160));
    await page.screenshot({ path: `${OUT}/${TAG}-04-legado-bloqueado.png` });
  } else {
    ok('legado ambíguo: botão disponível para conferir', false);
  }
  await fechar();

  // 3) Curral: abate cancelável pelo comando; sem Excluir
  await abrirArea('CURRAL 1');
  t = await txt();
  const excluir = await dlg().getByRole('button', { name: 'Excluir' }).count();
  ok('curral: sem botão Excluir; abate com "Cancelar movimentação"', excluir === 0 && new RegExp(`ABATE ${TAG}`).test(t), t.slice(0, 140));
  await tocar(itemDoLote(`ABATE ${TAG}`).getByRole('button', { name: 'Cancelar movimentação' }));
  await page.getByText('0 → 30 cab').first().waitFor({ timeout: 15000 });
  await page.getByLabel('Motivo do cancelamento').fill('Abate sintético lançado em duplicidade');
  await tocar(page.getByRole('button', { name: 'Confirmar cancelamento' }));
  await page.waitForTimeout(3000);
  await page.screenshot({ path: `${OUT}/${TAG}-05-curral-abate-cancelado.png` });
  t = await txt();
  ok('curral: abate aparece Cancelado com motivo', /Cancelado por demo\.operador — Motivo: Abate sintético/.test(t), t.slice(0, 160));
  await fechar();

  ok('nenhum DELETE/PATCH de movimento/suplementação no navegador', perigosos.length === 0, perigosos.join(', '));
  ok('cancelamentos foram POST no comando do servidor (2)', cancelamentos.length === 2, cancelamentos.length);
  ok('sem erro de página', erros.length === 0, erros.join(' ; '));
  console.log(res.join('\n'));
  await browser.close();
})();
